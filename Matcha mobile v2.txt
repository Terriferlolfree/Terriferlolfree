-- Helper to make any GUI element draggable (Mobile & PC)
local function MakeDraggable(gui)
    local dragging, dragStart, startPos
    gui.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = gui.Position
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            gui.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)

    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
end

function glow.setcolor(newcolor)
    glow.currentcolor = newcolor
    for i = 1, #glow.registry do
        local g = glow.registry[i]
        if g and g:IsA("ImageLabel") then
            g.ImageColor3 = newcolor
        end
    end
end

    local inst = Instance.new(class)

local CurrentTarget = nil
local TargetLockConnection = nil

local function GetClosestTarget()
    local maxDist = math.huge
    local center = Vector2.new(CurrentCamera.ViewportSize.X / 2, CurrentCamera.ViewportSize.Y / 2)
    local closestPlayer = nil

    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character then
            local root = p.Character:FindFirstChild("HumanoidRootPart")
            if root then
                local pos = root.Position + Vector3.new(Settings.PREDICTION.horizontalPrediction, Settings.PREDICTION.verticalPrediction, 0)
                local screenPos, onScreen = CurrentCamera:WorldToViewportPoint(pos)
                if onScreen then
                    local dist = (center - Vector2.new(screenPos.X, screenPos.Y)).Magnitude
                    if dist < maxDist then
                        closestPlayer = p
                        maxDist = dist
                    end
                end
            end
        end
    end
    return closestPlayer
end

local function ToggleTargetLock()
    Settings.AIMBOT.targetaim = not Settings.AIMBOT.targetaim

    if Settings.AIMBOT.targetaim then
        CurrentTarget = GetClosestTarget()
        if CurrentTarget and CurrentTarget.Character then
            FloatingLockButton.Text = "🔒 Locked"
            FloatingLockButton.TextColor3 = Color3.fromRGB(240, 110, 180)

            if TargetLockConnection then TargetLockConnection:Disconnect() end

            TargetLockConnection = RunService.Heartbeat:Connect(function()
                if Settings.AIMBOT.targetaim and CurrentTarget and CurrentTarget.Character and CurrentTarget.Character:FindFirstChild("HumanoidRootPart") then
                    local char = CurrentTarget.Character
                    local hum = char:FindFirstChildOfClass("Humanoid")
                    local aimPosition

                    if hum then
                        if hum.FloorMaterial ~= Enum.Material.Air then
                            if hum:GetState() ~= Enum.HumanoidStateType.Freefall and hum:GetState() ~= Enum.HumanoidStateType.Jumping then
                                aimPosition = char.Head.Position
                            else
                                aimPosition = char.Head.Position + Vector3.new(0, Settings.PREDICTION.jumpoffset, 0)
                            end
                        else
                            local part = char:FindFirstChild(Settings.AIMBOT.aimpart)
                            aimPosition = part and (part.Position + Vector3.new(0, Settings.PREDICTION.jumpoffset, 0)) or (char.Head.Position + Vector3.new(0, Settings.PREDICTION.jumpoffset, 0))
                        end
                    end

                    if aimPosition then
                        local targetCFrame = CFrame.new(CurrentCamera.CFrame.Position, aimPosition + Vector3.new(
                            char.HumanoidRootPart.Velocity.X * Settings.PREDICTION.horizontalPrediction,
                            char.HumanoidRootPart.Velocity.Y * Settings.PREDICTION.verticalPrediction,
                            char.HumanoidRootPart.Velocity.Z * Settings.PREDICTION.horizontalPrediction

                            end

            Settings.AIMBOT.targetaim = false
            FloatingLockButton.Text = "🔓 Lock"
            FloatingLockButton.TextColor3 = Color3.fromRGB(255, 255, 255)
        end
    else
        FloatingLockButton.Text = "🔓 Lock"
        FloatingLockButton.TextColor3 = Color3.fromRGB(255, 255, 255)
        if TargetLockConnection then
            TargetLockConnection:Disconnect()
            TargetLockConnection = nil
        end
        CurrentTarget = nil
    end
end

local function GetSilentTarget()
    local radius = Settings.SILENTAIM.FOVRadius
    local center = Vector2.new(CurrentCamera.ViewportSize.X / 2, CurrentCamera.ViewportSize.Y / 2)
    local target = nil

    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild(Settings.SILENTAIM.TargetPart) then
            if not Settings.SILENTAIM.TeamCheck or p.Team ~= LocalPlayer.Team then
                local screenPos, onScreen = CurrentCamera:WorldToViewportPoint(p.Character[Settings.SILENTAIM.TargetPart].Position)
                if onScreen then
                    local mag = (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude
                    if mag < radius then
                        target = p
                        radius = mag
                    end
                end
            end
        end
    end
    return target
end

local function RedirectVector(v)
    if typeof(v) == "Vector3" and CurrentTarget and CurrentTarget.Character and CurrentTarget.Character:FindFirstChild("HumanoidRootPart") then
        local hrp = CurrentTarget.Character.HumanoidRootPart
        local predicted = hrp.Position + Vector3.new(hrp.Velocity.X * Settings.PREDICTION.horizontalPrediction, hrp.Velocity.Y * Settings.PREDICTION.verticalPrediction, hrp.Velocity.Z * Settings.PREDICTION.horizontalPrediction)
        if predicted.Y < 0 and Settings.AIMBOT.antigroundshoot then
            predicted = Vector3.new(predicted.X, 0, predicted.Z)
        end
        return predicted
    end
    return v
end

rawMeta.__namecall = newcclosure(function(self, ...)
    local args = {...}
    local method = getnamecallmethod()
    if Settings.AIMBOT.targetaim and CurrentTarget and method == "FireServer" then
        for i, arg in ipairs(args) do
            if typeof(arg) == "Vector3" then
                args[i] = RedirectVector(arg)
            end
        end
    end
    return oldNamecall(self, unpack(args))
end)

local flyBodyGyro, flyBodyVel
local function ToggleFly(state)
    Settings.PLAYER.fly = state
    local char = LocalPlayer.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    local root = char.PrimaryPart or char:FindFirstChild("HumanoidRootPart")

    if Settings.PLAYER.fly and hum and root then
        hum.PlatformStand = true
        flyBodyGyro = Instance.new("BodyGyro", root)
        flyBodyGyro.P = 90000
        flyBodyGyro.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
        flyBodyGyro.CFrame = CurrentCamera.CFrame

        flyBodyVel = Instance.new("BodyVelocity", root)
        flyBodyVel.MaxForce = Vector3.new(9e9, 9e9, 9e9)

        task.spawn(function()
            while Settings.PLAYER.fly and char and hum.Parent do
                flyBodyGyro.CFrame = CurrentCamera.CFrame
                local moveDir = hum.MoveDirection
                local vel = moveDir.Magnitude > 0 and (CurrentCamera.CFrame.LookVector * moveDir.Magnitude) or Vector3.new(0, 0, 0)
                local vertical = 0
                if UserInputService:IsKeyDown(Enum.KeyCode.Space) then vertical = Settings.PLAYER.flyspeed end
                if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then vertical = -Settings.PLAYER.flyspeed end

                flyBodyVel.Velocity = vel * Settings.PLAYER.flyspeed + Vector3.new(0, vertical, 0)
                hum:ChangeState(Enum.HumanoidStateType.Freefall)
                RunService.RenderStepped:Wait()
            end
            if flyBodyGyro then flyBodyGyro:Destroy() end
            if flyBodyVel then flyBodyVel:Destroy() end
            if hum then hum.PlatformStand = false end
        end)
    else
        if flyBodyGyro then flyBodyGyro:Destroy() end
        if flyBodyVel then flyBodyVel:Destroy() end
        if hum then hum.PlatformStand = false end
    end
end

local function CreateESP(player)
    if player == LocalPlayer then return end
    local boxLines = {
        top = Drawing.new("Line"),
        bottom = Drawing.new("Line"),
        left = Drawing.new("Line"),
        right = Drawing.new("Line")
    }
    for _, l in pairs(boxLines) do
        l.Thickness = 1
        l.Color = Settings.ESP.Box_Color
        l.Visible = false
    end

    local nameText = Drawing.new("Text")
    nameText.Size = 13
    nameText.Center = true
    nameText.Outline = true
    nameText.Visible = false

    local distText = Drawing.new("Text")
    distText.Size = 12
    distText.Center = true
    distText.Outline = true
    distText.Visible = false

    local hpText = Drawing.new("Text")
    hpText.Size = 12
    hpText.Center = true
    hpText.Outline = true
    hpText.Color = Color3.fromRGB(0, 255, 120)
    hpText.Visible = false

    local conn
    conn = RunService.RenderStepped:Connect(function()
        if Settings.ESP.Enabled and player.Character and player.Character:FindFirstChild("HumanoidRootPart") and player.Character:FindFirstChild("Head") then
            local hrp = player.Character.HumanoidRootPart
            local screenPos, onScreen = CurrentCamera:WorldToViewportPoint(hrp.Position)

            if onScreen then
                local headPos = CurrentCamera:WorldToViewportPoint(player.Character.Head.Position)
                local feetPos = CurrentCamera:WorldToViewportPoint(hrp.Position - Vector3.new(0, 3, 0))
                local height = (headPos - feetPos).Magnitude * 1.5
                local width = height / 2

                if Settings.ESP.BoxEsp then
                    boxLines.top.From = Vector2.new(screenPos.X - width / 2, screenPos.Y - height / 2)
                    boxLines.top.To = Vector2.new(screenPos.X + width / 2, screenPos.Y - height / 2)
                    boxLines.bottom.From = Vector2.new(screenPos.X - width / 2, screenPos.Y + height / 2)
                    boxLines.bottom.To = Vector2.new(screenPos.X + width / 2, screenPos.Y + height / 2)
                    boxLines.left.From = Vector2.new(screenPos.X - width / 2, screenPos.Y - height / 2)
                    boxLines.left.To = Vector2.new(screenPos.X - width / 2, screenPos.Y + height / 2)
                    boxLines.right.From = Vector2.new(screenPos.X + width / 2, screenPos.Y - height / 2)
                    boxLines.right.To = Vector2.new(screenPos.X + width / 2, screenPos.Y + height / 2)

                    for _, l in pairs(boxLines) do
                        l.Color = Settings.ESP.Box_Color
                        l.Visible = true
                    end
                else
                    for _, l in pairs(boxLines) do l.Visible = false end
                end

                if Settings.ESP.NameEsp then
                    nameText.Position = Vector2.new(screenPos.X, screenPos.Y - height / 2 - 16)
                    nameText.Text = player.DisplayName .. " (@" .. player.Name .. ")"
                    nameText.Color = Settings.ESP.Name_Color
                    nameText.Visible = true
                else
                    nameText.Visible = false
                end

                if Settings.ESP.DistanceEsp then
                    local dist = (hrp.Position - CurrentCamera.CFrame.Position).Magnitude
                    distText.Position = Vector2.new(screenPos.X, screenPos.Y + height / 2 + 4)
                    distText.Text = string.format("%.1f studs", dist)
                    distText.Color = Settings.ESP.Distance_Color
                    distText.Visible = true
                else
                    distText.Visible = false
                end

                if Settings.ESP.hpesp and player.Character:FindFirstChildOfClass("Humanoid") then
                    hpText.Position = Vector2.new(screenPos.X, screenPos.Y - height / 2 - 28)
                    hpText.Text = string.format("HP: %d", player.Character:FindFirstChildOfClass("Humanoid").Health)
                    hpText.Visible = true
                else
                    hpText.Visible = false
                end
            else
                for _, l in pairs(boxLines) do l.Visible = false end
                nameText.Visible = false
                distText.Visible = false
                hpText.Visible = false
            end
        else
            for _, l in pairs(boxLines) do l.Visible = false end
            nameText.Visible = false
            distText.Visible = false
            hpText.Visible = false
            if not player.Parent then
                conn:Disconnect()
                for _, l in pairs(boxLines) do l:Remove() end
                nameText:Remove()
                distText:Remove()
                hpText:Remove()
            end
        end
    end)
end

local function FlingTarget(p)
    if not p or not p.Character or not LocalPlayer.Character then return end
    local myRoot = LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    local tRoot = p.Character:FindFirstChild("HumanoidRootPart")
    if not myRoot or not tRoot then return end

    local bVel = Instance.new("BodyVelocity")
    bVel.Velocity = Vector3.new(9e8, 9e8, 9e8)
    bVel.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
    bVel.Parent = myRoot

    local oldPos = myRoot.CFrame
    for i = 1, 30 do
        myRoot.CFrame = tRoot.CFrame * CFrame.Angles(math.rad(math.random(0, 360)), 0, 0)
        task.wait()
    end
    bVel:Destroy()
    myRoot.CFrame = oldPos
end
